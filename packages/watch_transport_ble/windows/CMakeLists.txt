cmake_minimum_required(VERSION 3.16)
project(watch_transport_ble LANGUAGES CXX)
set(PLUGIN_NAME "watch_transport_ble_plugin")
if(TARGET flutter)
add_library(${PLUGIN_NAME} SHARED "watch_transport_ble_plugin.cpp" "bluetooth_address.cpp")
apply_standard_settings(${PLUGIN_NAME})
target_compile_features(${PLUGIN_NAME} PRIVATE cxx_std_17)
target_compile_definitions(${PLUGIN_NAME} PRIVATE FLUTTER_PLUGIN_IMPL NOMINMAX)
if(MSVC)
  target_compile_options(${PLUGIN_NAME} PRIVATE /await)
endif()
target_include_directories(${PLUGIN_NAME} INTERFACE "${CMAKE_CURRENT_SOURCE_DIR}/include")
target_link_libraries(${PLUGIN_NAME} PRIVATE flutter flutter_wrapper_plugin windowsapp)
set(watch_transport_ble_bundled_libraries "" PARENT_SCOPE)
endif()
option(BUILD_TESTING "Build BLE native unit tests" OFF)
if(BUILD_TESTING)
  enable_testing()
  add_executable(watch_ble_address_test "tests/bluetooth_address_test.cpp" "bluetooth_address.cpp")
  target_compile_features(watch_ble_address_test PRIVATE cxx_std_17)
  target_include_directories(watch_ble_address_test PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
  add_test(NAME watch_ble_address COMMAND watch_ble_address_test)
endif()
